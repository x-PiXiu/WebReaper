# 发布系统模块分析

> 多平台内容发布的核心业务

## 1. 模块概述

发布系统模块支持内容一键发布到多个平台（知乎、小红书、抖音等），提供半自动和全自动两种发布模式。

**业务目标**：
- 支持多平台内容发布（知乎/小红书/抖音）
- 提供半自动（预填 URL）和全自动（RPA）两种模式
- 账号池管理（最优账号选择）
- 发布效果追踪（互动数据回读）

```go
func (uc *AccountUseCase) StartQRLogin(ctx context.Context, platform, method string) (sessionID string, err error) {
    // 1. 检查扫码登录是否启用
    if uc.qrLogin == nil {
        return "", fmt.Errorf("扫码登录未启用")
    }
    
    // 2. 启动扫码登录
    return uc.qrLogin.StartLogin(ctx, platform, method)
}

func (uc *AccountUseCase) PollQRLogin(ctx context.Context, tenantID, sessionID, platform, method string) (PollQRLoginResult, error) {
    // 1. 轮询扫码状态
    result, err := uc.qrLogin.PollStatus(ctx, sessionID)
    if err != nil {
        return PollQRLoginResult{Status: "error"}, err
    }
    
    // 2. 登录成功 → 处理 cookie
    if result.Status == "success" {
        // cookie 加密
        encCookie, err := uc.vault.Encrypt(result.Cookie)
        if err != nil {
            return PollQRLoginResult{Status: "error"}, err
        }
        
        // 创建账号
        account := entity.Account{
            ID:          generateID(),
            TenantID:    tenantID,
            Platform:    platform,
            DisplayName: platformDisplayName(platform),
            Health:      entity.AccountHealthActive,
            LoginMethod: method,
            CookieEnc:   encCookie,
            ExpiresAt:   result.ExpiresAt,
            BoundAt:     time.Now(),
            LastUsedAt:  time.Now(),
        }
        
        // 账号落库
        if err := uc.accountRepo.Save(ctx, account); err != nil {
            return PollQRLoginResult{Status: "error"}, err
        }
        
        return PollQRLoginResult{
            Status:      "success",
            AccountID:   account.ID,
            AccountName: account.DisplayName,
            ExpiresAt:   account.ExpiresAt,
        }, nil
    }
    
    return PollQRLoginResult{
        Status:  result.Status,
        QRImage: result.QRImage,
    }, nil
}
```

#### 3.1.2 OAuth 授权绑定

```go
func (uc *AccountUseCase) BuildOAuthURL(tenantID, userID string) (string, error) {
    // 1. 检查 OAuth 是否配置
    if uc.oauth == nil || uc.stateCodec == nil {
        return "", fmt.Errorf("官方 OAuth 授权未配置")
    }
    
    // 2. 生成 state（签名）
    state := uc.stateCodec.SignState(tenantID + "|" + userID)
    
    // 3. 生成授权页地址
    return uc.oauth.ConnectURL(state), nil
}

func (uc *AccountUseCase) HandleOAuthCallback(ctx context.Context, code, state string) (string, string, error) {
    // 1. 验 state
    payload, err := uc.stateCodec.VerifyState(state)
    if err != nil {
        return "", "", fmt.Errorf("授权状态校验失败")
    }
    tenantID, _, _ := strings.Cut(payload, "|")
    
    // 2. code 换 token
    token, err := uc.oauth.ExchangeCode(ctx, code)
    if err != nil {
        return "", "", fmt.Errorf("授权码换取 token 失败")
    }
    
    // 3. token 加密
    encAccess, _ := uc.vault.Encrypt(token.AccessToken)
    encRefresh, _ := uc.vault.Encrypt(token.RefreshToken)
    
    // 4. 拉用户信息
    displayName := platformDisplayName("douyin")
    if info, err := uc.oauth.UserInfo(ctx, token.AccessToken, token.OpenID); err == nil {
        if info.Nickname != "" {
            displayName = info.Nickname
        }
    }
    
    // 5. 创建账号
    account := entity.Account{
        ID:               generateID(),
        TenantID:         tenantID,
        Platform:         "douyin",
        DisplayName:      displayName,
        Health:           entity.AccountHealthActive,
        LoginMethod:      "douyin",
        AuthType:         entity.AccountAuthOAuth,
        AccessTokenEnc:   encAccess,
        RefreshTokenEnc:  encRefresh,
        OpenID:           token.OpenID,
        ExpiresAt:        time.Now().Add(time.Duration(token.ExpiresIn) * time.Second),
        BoundAt:          time.Now(),
        LastUsedAt:       time.Now(),
    }
    
    // 6. 账号落库
    if err := uc.accountRepo.Save(ctx, account); err != nil {
        return "", "", err
    }
    
    return account.ID, account.DisplayName, nil
}
```

### 3.2 发布流程

#### 3.2.1 半自动发布

```go
func (uc *PublishUseCase) PublishSemiAuto(ctx context.Context, input PublishInput) (PublishOutput, error) {
    // 1. 获取发布通道
    channel, err := uc.registry.Get(input.Platform)
    if err != nil {
        return PublishOutput{}, err
    }
    
    // 2. 借用账号
    account, err := uc.borrowAccount(ctx, input.TenantID, input.Platform)
    if err != nil {
        return PublishOutput{}, err
    }
    
    // 3. 能力校验
    if err := uc.validateCapability(channel, input.ContentType, input); err != nil {
        return PublishOutput{}, err
    }
    
    // 4. 解密 cookie
    cookie, err := uc.vault.Decrypt(account.CookieEnc)
    if err != nil {
        return PublishOutput{}, err
    }
    
    // 5. 生成预填 URL
    publishURL, err := channel.BuildPublishURL(ctx, port.PublishChannelInput{
        Cookie:      cookie,
        Title:       input.Title,
        Content:     input.Content,
        MediaURLs:   input.MediaURLs,
        ContentType: input.ContentType,
    })
    if err != nil {
        return PublishOutput{}, err
    }
    
    // 6. 记录发布任务
    job := entity.PublishJob{
        ID:          generateID(),
        TenantID:    input.TenantID,
        BrandID:     input.BrandID,
        AccountID:   account.ID,
        Platform:    input.Platform,
        ContentType: input.ContentType,
        Title:       input.Title,
        Status:      "pending",
        PublishURL:  publishURL,
        CreatedAt:   time.Now(),
    }
    
    if err := uc.jobRepo.Save(ctx, job); err != nil {
        return PublishOutput{}, err
    }
    
    // 7. 更新账号最后使用时间
    _ = uc.accountRepo.UpdateLastUsedAt(ctx, account.ID)
    
    return PublishOutput{
        JobID:      job.ID,
        PublishURL: publishURL,
        Platform:   input.Platform,
    }, nil
}
```

#### 3.2.2 全自动发布

```go
func (uc *PublishUseCase) PublishAuto(ctx context.Context, input PublishInput) (PublishOutput, error) {
    // 1. 获取发布通道
    channel, err := uc.registry.Get(input.Platform)
    if err != nil {
        return PublishOutput{}, err
    }
    
    // 2. 检查是否支持全自动
    autoChannel, ok := channel.(port.AutoPublishChannel)
    if !ok {
        return PublishOutput{}, fmt.Errorf("该平台不支持全自动发布")
    }
    
    // 3. 选择最优账号
    account, err := uc.accountPool.GetBest(ctx, input.TenantID, input.Platform)
    if err != nil {
        return PublishOutput{}, err
    }
    
    // 4. 能力校验
    if err := uc.validateCapability(channel, input.ContentType, input); err != nil {
        return PublishOutput{}, err
    }
    
    // 5. 解密凭证
    cred, err := uc.resolver.Resolve(ctx, account)
    if err != nil {
        return PublishOutput{}, err
    }
    
    // 6. 全自动发布
    result, err := autoChannel.AutoPublish(ctx, port.AutoPublishInput{
        Credential:  cred,
        Title:       input.Title,
        Content:     input.Content,
        MediaURLs:   input.MediaURLs,
        ContentType: input.ContentType,
    })
    if err != nil {
        return PublishOutput{}, err
    }
    
    // 7. 记录发布任务
    job := entity.PublishJob{
        ID:            generateID(),
        TenantID:      input.TenantID,
        BrandID:       input.BrandID,
        AccountID:     account.ID,
        Platform:      input.Platform,
        ContentType:   input.ContentType,
        Title:         input.Title,
        Status:        "published",
        PublishedURL:  result.PublishedURL,
        PublishedAt:   time.Now(),
        CreatedAt:     time.Now(),
    }
    
    if err := uc.jobRepo.Save(ctx, job); err != nil {
        return PublishOutput{}, err
    }
    
    // 8. 触发效果追踪
    if uc.monitorTrigger != nil {
        _ = uc.monitorTrigger.Trigger(ctx, input.TenantID, input.BrandID)
    }
    
    // 9. 更新账号最后使用时间
    _ = uc.accountRepo.UpdateLastUsedAt(ctx, account.ID)
    
    return PublishOutput{
        JobID:        job.ID,
        PublishedURL: result.PublishedURL,
        Platform:     input.Platform,
    }, nil
}
```

### 3.3 账号池调度

```go
func (p *GormAccountPool) GetBest(ctx context.Context, tenantID, platform string) (entity.Account, error) {
    // 1. 查询可用账号
    accounts, err := p.accountRepo.ListByPlatform(ctx, tenantID, platform)
    if err != nil {
        return entity.Account{}, err
    }
    
    // 2. 过滤健康账号
    var healthy []entity.Account
    for _, acc := range accounts {
        if acc.Health == entity.AccountHealthActive && acc.ExpiresAt.After(time.Now()) {
            healthy = append(healthy, acc)
        }
    }
    
    if len(healthy) == 0 {
        return entity.Account{}, fmt.Errorf("没有可用的健康账号")
    }
    
    // 3. 最优选择：最久未使用优先
    sort.Slice(healthy, func(i, j int) bool {
        return healthy[i].LastUsedAt.Before(healthy[j].LastUsedAt)
    })
    
    return healthy[0], nil
}
```

```go
func (uc *PublishUseCase) RefreshMetrics(ctx context.Context, tenantID string) error {
    // 1. 查询已发布的任务
    jobs, err := uc.jobRepo.ListByStatus(ctx, tenantID, "published")
    if err != nil {
        return err
    }
    
    // 2. 逐个刷新互动数据
    for _, job := range jobs {
        if uc.socialSearch == nil {
            continue
        }
        
        // 3. 站内搜索取详情
        detail, err := uc.socialSearch.GetDetail(ctx, job.PublishedURL)
        if err != nil {
            continue
        }
        
        // 4. 快照落库
        metric := entity.VideoMetric{
            ID:        generateID(),
            JobID:     job.ID,
            TenantID:  tenantID,
            Views:     detail.Views,
            Likes:     detail.Likes,
            Comments:  detail.Comments,
            Shares:    detail.Shares,
            ProbedAt:  time.Now(),
        }
        
        if uc.metricRepo != nil {
            _ = uc.metricRepo.Save(ctx, metric)
        }
    }
    
    return nil
}
```

## 5. 通道轴设计

### 5.1 TransportRegistry（传输层注册表）

```go
type TransportRegistry struct {
    transports map[string]port.Transport
}

func (r *TransportRegistry) Register(t port.Transport) {
    r.transports[t.Name()] = t
}

func (r *TransportRegistry) Get(name string) (port.Transport, error) {
    t, ok := r.transports[name]
    if !ok {
        return nil, fmt.Errorf("transport %s not found", name)
    }
    return t, nil
}
```

```go
func (uc *PublishUseCase) publishWithFallback(ctx context.Context, input PublishInput) (PublishOutput, error) {
    // 1. 尝试 RPA 传输
    if uc.transports != nil {
        rpa, err := uc.transports.Get("rpa")
        if err == nil {
            result, err := rpa.Publish(ctx, TransportInput{...})
            if err == nil {
                return result, nil
            }
        }
    }
    
    // 2. 降级到链接传输
    if uc.transports != nil {
        link, err := uc.transports.Get("link")
        if err == nil {
            result, err := link.Publish(ctx, TransportInput{...})
            if err == nil {
                return result, nil
            }
        }
    }
    
    return PublishOutput{}, fmt.Errorf("所有传输通道都失败")
}
```

```go
    // 1. 查询所有账号
    accounts, err := t.accountUC.ListAll(ctx)

        return err
```

### 9.2 工厂模式（Factory Pattern）

**通道注册表**：
```go
type ChannelRegistry struct {
    channels map[string]port.PublishChannel
}

func (r *ChannelRegistry) Register(channel port.PublishChannel) {
    r.channels[channel.Platform()] = channel
}

func (r *ChannelRegistry) Get(platform string) (port.PublishChannel, error) {
    channel, ok := r.channels[platform]
    if !ok {
        return nil, fmt.Errorf("channel %s not found", platform)
    }
    return channel, nil
}
```

### 9.3 装饰器模式（Decorator Pattern）

**凭证解析装饰器**：
```go
type VaultCredentialResolver struct {
    vault       port.CookieVault
    accountRepo port.AccountRepository
}

func (r *VaultCredentialResolver) Resolve(ctx context.Context, account entity.Account) (port.Credential, error) {
    // 根据 AuthType 解析凭证
    switch account.AuthType {
    case entity.AccountAuthCookie:
        cookie, err := r.vault.Decrypt(account.CookieEnc)
        if err != nil {
            return port.Credential{}, err
        }
        return port.Credential{Cookie: cookie}, nil
    case entity.AccountAuthOAuth:
        accessToken, err := r.vault.Decrypt(account.AccessTokenEnc)
        if err != nil {
            return port.Credential{}, err
        }
        return port.Credential{AccessToken: accessToken}, nil
    }
    return port.Credential{}, fmt.Errorf("unsupported auth type")
}
```

发布系统模块的设计体现了整洁架构的核心思想：
