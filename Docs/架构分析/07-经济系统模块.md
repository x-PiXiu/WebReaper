# 经济系统（套餐/计费）模块分析

> SaaS 级的套餐/订阅/计费管理

## 1. 模块概述

经济系统模块是 WebReaper 的**商业闭环核心**，提供完整的套餐/订阅/计费/配额管理。

**业务目标**：
- 套餐管理（free/basic/pro/enterprise）
- 订阅管理（月付/年付）
- 配额检查（按场景限额）
- 用量计量（token 级成本追踪）
- 支付集成（ZPay 真实收款）

### 3.2 订阅管理

#### 3.2.1 创建订阅

```go
func (uc *BillingUseCase) CreateSubscription(ctx context.Context, tenantID, planID string) (entity.Order, error) {
    // 1. 校验套餐有效性
    plan, err := uc.planRepo.FindByID(ctx, planID)
    if err != nil {
        return entity.Order{}, err
    }
    if plan.Status != entity.PlanStatusActive {
        return entity.Order{}, fmt.Errorf("套餐已下架")
    }
    
    // 2. 创建订单
    order := entity.Order{
        ID:          generateID(),
        TenantID:    tenantID,
        PlanID:      planID,
        AmountCents: plan.PriceCents,
        Status:      entity.OrderStatusPending,
        CreatedAt:   time.Now(),
    }
    
    if err := uc.orderRepo.Save(ctx, order); err != nil {
        return entity.Order{}, err
    }
    
    // 3. 调用支付网关（可选）
    if uc.payment != nil {
        resp, err := uc.payment.CreatePayment(ctx, port.PaymentRequest{
            OrderID:     order.ID,
            AmountCents: order.AmountCents,
            Description: fmt.Sprintf("订阅 %s", plan.Name),
        })
        if err != nil {
            return entity.Order{}, err
        }
        
        order.PaymentID = resp.PaymentID
        order.PayURL = resp.PayURL
        _ = uc.orderRepo.Save(ctx, order)
    }
    
    return order, nil
}
```

#### 3.2.2 支付回调处理

```go
func (uc *BillingUseCase) HandleCallback(ctx context.Context, params map[string]string) (string, error) {
    // 1. 验签
    result, err := uc.payment.VerifyCallback(ctx, params)
    if err != nil {
        return "fail", err
    }
    
    // 2. 查订单
    order, err := uc.orderRepo.FindByID(ctx, result.OutTradeNo)
    if err != nil {
        return "fail", err
    }
    
    // 3. 校验金额
    if order.AmountCents != result.AmountCents {
        return "fail", fmt.Errorf("金额不匹配")
    }
    
    // 4. 订单状态更新 + 订阅开通（原子操作）
    if uc.closureWriter != nil {
        err = uc.closureWriter.WritePaymentClosure(ctx, order.ID, result.PaymentID, func(ctx context.Context) error {
            // 开通订阅
            sub := entity.Subscription{
                ID:          generateID(),
                TenantID:    order.TenantID,
                PlanID:      order.PlanID,
                Status:      entity.SubscriptionStatusActive,
                PeriodStart: time.Now(),
                PeriodEnd:   time.Now().AddDate(0, 1, 0),  // 一个月
                CreatedAt:   time.Now(),
            }
            return uc.subRepo.Save(ctx, sub)
        })
    } else {
        // 降级：两段写
        order.Status = entity.OrderStatusPaid
        order.PaidAt = time.Now()
        if err := uc.orderRepo.Save(ctx, order); err != nil {
            return "fail", err
        }
        
        sub := entity.Subscription{
            ID:          generateID(),
            TenantID:    order.TenantID,
            PlanID:      order.PlanID,
            Status:      entity.SubscriptionStatusActive,
            PeriodStart: time.Now(),
            PeriodEnd:   time.Now().AddDate(0, 1, 0),
            CreatedAt:   time.Now(),
        }
        if err := uc.subRepo.Save(ctx, sub); err != nil {
            return "fail", err
        }
    }
    
    return "success", nil
}
```

### 3.3 配额检查

#### 3.3.1 配额检查门

```go
func (g *Gate) Check(ctx context.Context, tenantID, scene string) error {
    // 1. 未启用配额检查
    if g == nil {
        return nil
    }
    
    // 2. admin 旁路不计配额
    if tenantID == "" {
        return nil
    }
    
    // 3. 解析套餐和计费周期
    plan, periodStart, err := g.resolvePlanAndPeriod(ctx, tenantID)
    if err != nil {
        return nil  // 取不到套餐信息：放行
    }
    
    // 4. 取配额上限
    limit := plan.QuotaFor(scene)
    if limit == -1 {
        return nil  // 无限配额
    }
    if limit == 0 {
        return pkg.ErrQuotaExceeded  // 套餐未开通此场景
    }
    
    // 5. 取当前周期已用量
    used, err := g.usageQueryer.CountSince(ctx, tenantID, scene, periodStart)
    if err != nil {
        return nil  // 用量查询失败：放行
    }
    
    // 6. 判断是否超限
    if used >= limit {
        return pkg.ErrQuotaExceeded
    }
    
    return nil
}
```

#### 3.3.2 配额进度条

```go
func (g *Gate) QuotaFor(ctx context.Context, tenantID, scene string) (limit, used int, err error) {
    if g == nil || tenantID == "" {
        return -1, 0, nil
    }
    
    plan, periodStart, err := g.resolvePlanAndPeriod(ctx, tenantID)
    if err != nil {
        return -1, 0, err
    }
    
    limit = plan.QuotaFor(scene)
    if limit == -1 {
        return -1, 0, nil  // 无限
    }
    
    used, qErr := g.usageQueryer.CountSince(ctx, tenantID, scene, periodStart)
    if qErr != nil {
        return limit, 0, nil
    }
    
    return limit, used, nil
}
```

#### 3.3.3 套餐解析

```go
func (g *Gate) resolvePlanAndPeriod(ctx context.Context, tenantID string) (entity.Plan, time.Time, error) {
    // 1. 订阅存在且有效 → 用订阅的套餐 + 计费周期
    if sub, err := g.subRepo.FindByTenant(ctx, tenantID); err == nil {
        now := time.Now()
        if sub.IsActive(now) {
            if plan, pErr := g.planRepo.FindByID(ctx, sub.PlanID); pErr == nil {
                return plan, sub.PeriodStart, nil
            }
        }
    }
    
    // 2. 无订阅或订阅失效 → 降级 free 套餐，周期=当月 1 日起
    plan, err := g.planRepo.FindByID(ctx, g.freePlanID)
    if err != nil {
        return entity.Plan{}, time.Time{}, err
    }
    
    return plan, startOfCurrentMonth(), nil
}
```

#### 3.5.3 ZPay 支付网关

```go
type ZPayGateway struct {
    config ZPayConfig
}

type ZPayConfig struct {
    PID       string
    Key       string
    NotifyURL string
    ReturnURL string
}

func (g *ZPayGateway) CreatePayment(ctx context.Context, req PaymentRequest) (PaymentResponse, error) {
    // 1. 构建支付参数
    params := map[string]string{
        "pid":          g.config.PID,
        "type":         "alipay",
        "out_trade_no": req.OrderID,
        "notify_url":   g.config.NotifyURL,
        "return_url":   g.config.ReturnURL,
        "name":         req.Description,
        "money":        fmt.Sprintf("%.2f", float64(req.AmountCents)/100),
    }
    
    // 2. 生成签名
    sign := g.sign(params)
    params["sign"] = sign
    params["sign_type"] = "MD5"
    
    // 3. 构建支付 URL
    payURL := "https://zpay.cn/submit?" + encodeParams(params)
    
    return PaymentResponse{
        PaymentID: req.OrderID,
        PayURL:    payURL,
    }, nil
}

func (g *ZPayGateway) VerifyCallback(ctx context.Context, params map[string]string) (CallbackResult, error) {
    // 1. 验签
    sign := params["sign"]
    delete(params, "sign")
    delete(params, "sign_type")
    
    if g.sign(params) != sign {
        return CallbackResult{}, fmt.Errorf("签名验证失败")
    }
    
    // 2. 提取回调数据
    return CallbackResult{
        OutTradeNo:  params["out_trade_no"],
        PaymentID:   params["trade_no"],
        AmountCents: parseAmount(params["money"]),
    }, nil
}
```

## 5. 成本分析

### 5.1 成本分析报表

```go
func (uc *BillingUseCase) CostReport(ctx context.Context, tenantID string, start, end time.Time) (CostReport, error) {
    // 1. 查询用量统计
    usageStats, err := uc.usageStats.TotalTokens(ctx, tenantID, start, end)
    if err != nil {
        return CostReport{}, err
    }
    
    // 2. 估算成本
    costEstimate, err := uc.usageStats.CostEstimate(ctx, tenantID, start, end, uc.perMTokenCents)
    if err != nil {
        return CostReport{}, err
    }
    
    // 3. 按引擎细分成本（P1-1）
    engineCosts := make(map[string]int64)
    if uc.llmCfgRepo != nil {
        configs, _ := uc.llmCfgRepo.ListAll(ctx)
        for _, config := range configs {
            if config.CostPerMTokenCents > 0 {
                engineTokens, _ := uc.usageStats.TotalTokensByEngine(ctx, tenantID, config.Name, start, end)
                engineCosts[config.Name] = engineTokens * int64(config.CostPerMTokenCents) / 1000000
            }
        }
    }
    
    return CostReport{
        TenantID:      tenantID,
        PeriodStart:   start,
        PeriodEnd:     end,
        TotalTokens:   usageStats,
        CostEstimate:  costEstimate,
        EngineCosts:   engineCosts,
    }, nil
}
```

### 5.2 收入报表

```go
func (uc *BillingUseCase) RevenueReport(ctx context.Context, start, end time.Time) (RevenueReport, error) {
    // 1. 查询已支付订单
    orders, err := uc.orderRepo.ListByStatus(ctx, entity.OrderStatusPaid, start, end)
    if err != nil {
        return RevenueReport{}, err
    }
    
    // 2. 按套餐统计
    planRevenue := make(map[string]int)
    for _, order := range orders {
        planRevenue[order.PlanID] += order.AmountCents
    }
    
    // 3. 计算总收入
    totalRevenue := 0
    for _, revenue := range planRevenue {
        totalRevenue += revenue
    }
    
    return RevenueReport{
        PeriodStart:  start,
        PeriodEnd:    end,
        TotalRevenue: totalRevenue,
        PlanRevenue:  planRevenue,
        OrderCount:   len(orders),
    }, nil
}
```


### 7.2 聚合关系

## 8. 支付闭环原子化

### 8.1 问题描述

**两段写的风险**：
```
1. 订单状态更新为 paid
2. 创建订阅
```
如果步骤 2 失败，会导致"已付款未开通"的中间态。

### 8.2 解决方案

**PaymentClosureWriter（支付闭环原子写入器）**：
```go
type PaymentClosureWriter interface {
    WritePaymentClosure(ctx context.Context, orderID, paymentID string, createSub func(ctx context.Context) error) error
}

// GormPaymentClosureWriter GORM 实现
type GormPaymentClosureWriter struct {
    db *gorm.DB
}

func (w *GormPaymentClosureWriter) WritePaymentClosure(ctx context.Context, orderID, paymentID string, createSub func(ctx context.Context) error) error {
    return w.db.WithContext(ctx).Transaction(func(tx *gorm.DB) error {
        // 1. 更新订单状态
        if err := tx.Model(&OrderPO{}).Where("id = ?", orderID).Updates(map[string]interface{}{
            "status":      entity.OrderStatusPaid,
            "payment_id":  paymentID,
            "paid_at":     time.Now(),
        }).Error; err != nil {
            return err
        }
        
        // 2. 创建订阅
        if err := createSub(ctx); err != nil {
            return err
        }
        
        return nil
    })
}
```

## 9. 总结

经济系统模块的设计体现了整洁架构的核心思想：
