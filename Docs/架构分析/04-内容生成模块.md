# 内容生成模块分析

> 统一生成任务协议：视频/图片/音频/数字人全量接入

## 1. 模块概述

内容生成模块是 WebReaper 的**核心能力之一**，提供统一的任务协议支持多种媒体类型生成。

**设计目标**：
- **统一协议**：所有端点共享同一任务模型（提交→task_id→轮询/回调→creations）
- **策略驱动**：端点差异通过 EndpointAdapter 策略实现
- **数据库驱动**：模型能力由 generation_specs 表管理，管理后台可热更新
- **双通道结果获取**：回调（秒级推送）+ 轮询（20s 兜底）

## 3. 核心业务流程

### 3.1 提交流程

```go
func (uc *GenerationUseCase) Submit(ctx context.Context, in SubmitInput) (entity.GenerationTask, error) {
    // 1. 配额检查
    if uc.quotaGate != nil {
        if err := uc.quotaGate.Check(ctx, in.TenantID, "generation"); err != nil {
            return entity.GenerationTask{}, err
        }
    }
    
    // 2. 获取端点适配器
    adapter, err := uc.registry.Get(ctx, in.SubType)
    if err != nil {
        return entity.GenerationTask{}, err
    }
    
    // 3. 模型自动选择（傻瓜式端点）
    model := in.Model
    if model == "" {
        if sel, ok := adapter.(port.ModelAutoSelector); ok {
            model = uc.pickModelFor(ctx, in.SubType, sel, in.Params)
        }
    }
    
    // 4. 获取模型能力
    cap, err := uc.registry.Capability(ctx, in.SubType, model)
    if err != nil {
        return entity.GenerationTask{}, err
    }
    
    // 5. 提示词翻译（@引用 → 上游参数格式）
    params, err := translateRefs(in.SubType, cap, in.Params, in.Refs)
    if err != nil {
        return entity.GenerationTask{}, err
    }
    
    // 6. 参数校验
    if err := adapter.Validate(ctx, cap, params); err != nil {
        return entity.GenerationTask{}, err
    }
    
    // 7. 防重复提交
    hash := paramsHash(in.SubType, model, params)
    if pending, pErr := uc.repo.FindPendingByHash(ctx, in.TenantID, hash); pErr == nil && len(pending) > 0 {
        return pending[0], nil
    }
    
    // 8. 创建任务
    task := entity.GenerationTask{
        ID:        fmt.Sprintf("gen-%d", now.UnixNano()),
        TenantID:  in.TenantID,
        BrandID:   in.BrandID,
        Type:      adapter.Category(),
        SubType:   in.SubType,
        Model:     model,
        Provider:  uc.provider.Name(),
        State:     entity.TaskStateCreated,
        ParamsHash: hash,
        CreatedAt: now,
    }
    
    // 9. 任务落库
    if err := uc.repo.Save(ctx, task); err != nil {
        return entity.GenerationTask{}, err
    }
    
    // 10. 构建请求体
    body, bErr := adapter.BuildRequest(ctx, model, params, task.Payload)
    if bErr != nil {
        task.State = entity.TaskStateFailed
        task.ErrMsg = "参数组装失败: " + bErr.Error()
        _ = uc.repo.Save(ctx, task)
        return task, bErr
    }
    
    // 11. 注入回调地址
    uc.injectCallbackURL(adapter, body)
    
    // 12. 本地媒体内联
    uc.inlineLocalMedia(ctx, body)
    
    // 13. 并发节流
    if uc.submitSem != nil {
        select {
        case uc.submitSem <- struct{}{}:
            defer func() { <-uc.submitSem }()
        case <-ctx.Done():
            return task, ctx.Err()
        }
    }
    
    // 14. 调用 provider.Submit()
    res, err := uc.provider.Submit(ctx, adapter.Endpoint(), body)
    if err != nil {
        task.State = entity.TaskStateFailed
        task.ErrMsg = fmt.Sprintf("提交失败: %v", err)
        _ = uc.repo.Save(ctx, task)
        return task, err
    }
    
    // 15. 更新任务状态
    task.ProviderTaskID = res.TaskID
    task.Credits = res.Credits
    uc.applySubmitResult(ctx, &task, adapter, res)
    _ = uc.repo.Save(ctx, task)
    
    // 16. 记录用量
    uc.recordUsage(ctx, task.TenantID, task.Model, res.Credits)
    
    return task, nil
}
```

### 3.2 查询流程

```go
func (uc *GenerationUseCase) Get(ctx context.Context, tenantID, taskID string) (entity.GenerationTask, error) {
    // 1. 从仓储读取任务
    task, err := uc.repo.FindByID(ctx, tenantID, taskID)
    if err != nil {
        return entity.GenerationTask{}, err
    }
    
    // 2. 判断是否终态
    if entity.IsTerminal(task.State) {
        return task, nil
    }
    
    // 3. 调用 provider.Query() 更新状态
    status, err := uc.provider.Query(ctx, task.ProviderTaskID)
    if err != nil {
        return task, nil  // 查询失败不阻塞，下次轮询重试
    }
    
    // 4. 应用状态更新
    uc.applyStatus(ctx, &task, status)
    _ = uc.repo.Save(ctx, task)
    
    return task, nil
}
```

### 3.3 回调处理流程

```go
func (uc *GenerationUseCase) HandleCallback(ctx context.Context, payload, providerTaskID string, status port.GenerationStatus) (entity.GenerationTask, error) {
    // 1. Nonce 判重
    if !uc.nonceStore.Seen(ctx, payload) {
        return entity.GenerationTask{}, nil  // 重复回调忽略
    }
    
    // 2. 任务查找（双路径：payload → provider_task_id）
    task, err := uc.repo.FindByID(ctx, "", payload)
    if err != nil {
        if providerTaskID != "" {
            task, err = uc.repo.FindByProviderTaskID(ctx, providerTaskID)
            if err != nil {
                return entity.GenerationTask{}, err
            }
        }
    }
    
    // 3. 终态幂等
    if entity.IsTerminal(task.State) {
        return task, nil
    }
    
    // 4. 回调到达标记
    task.CallbackReceived = true
    now := time.Now()
    task.CallbackAt = &now
    
    // 5. 应用状态更新
    uc.applyStatus(ctx, &task, status)
    _ = uc.repo.Save(ctx, task)
    
    return task, nil
}
```

### 5.2 数据库驱动管理

**generation_specs 表**：
```sql
CREATE TABLE generation_specs (
    sub_type VARCHAR(50) NOT NULL,
    model VARCHAR(100) NOT NULL,
    endpoint VARCHAR(200) NOT NULL,
    enabled BOOLEAN DEFAULT TRUE,
    capabilities_json TEXT,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    PRIMARY KEY (sub_type, model)
);
```

**管理逻辑**：
1. **首次启动 Seed**：代码默认能力表写入 DB
2. **查询优先级**：DB 优先（30s TTL 缓存）→ 代码默认回退
3. **管理后台编辑**：直接修改 DB 记录，30s 内生效
4. **删除行**：恢复代码默认（出厂值回退）
5. **新增模型**：管理后台直接插入一行

### 5.3 热更新机制

```go
func (r *Registry) specs(ctx context.Context) []entity.GenerationSpec {
    r.mu.Lock()
    defer r.mu.Unlock()
    
    if r.specRepo != nil {
        // 30s TTL 缓存
        if r.cacheList == nil || time.Since(r.cachedAt) > 30*time.Second {
            if list, err := r.specRepo.ListAll(ctx); err == nil {
                r.cacheList = list
                r.cachedAt = time.Now()
            }
        }
        if r.cacheList != nil {
            return r.cacheList
        }
    }
    
    // 纯代码模式：默认表转 spec 列表
    var out []entity.GenerationSpec
    for subType, caps := range r.defaultCaps {
        for _, c := range caps {
            out = append(out, entity.GenerationSpec{
                SubType: subType, Model: c.Model, Endpoint: endpointOf(subType), Enabled: true,
            })
        }
    }
    return out
}
```

// scheduledtask/generation_poll.go
type GenerationPollTask struct {
    genUC *generation.GenerationUseCase
    log   port.Logger
}

func (t *GenerationPollTask) Run(ctx context.Context) error {
    // 1. 查询未终态任务
    tasks, err := t.genUC.ListPending(ctx)
    if err != nil {
        return err
    }
    
    // 2. 逐个查询状态
    for _, task := range tasks {
        if task.CallbackReceived {
            continue  // 回调已到达，跳过轮询
        }
        _, _ = t.genUC.Get(ctx, task.TenantID, task.ID)
    }
    
    return nil
}

type memoryNonceStore struct {
    nonces map[string]time.Time
}

func (s *memoryNonceStore) Seen(_ context.Context, nonce string) bool {
    now := time.Now()
    if t, ok := s.nonces[nonce]; ok && now.Sub(t) < 5*time.Minute {
        return false  // 5 分钟内重复
    }
    s.nonces[nonce] = now
    return true
}

func (uc *GenerationUseCase) inlineLocalMedia(ctx context.Context, body map[string]any) {
    if uc.asset == nil {
        return
    }
    
    // 数组字段
    for _, key := range []string{"images", "videos"} {
        if arr, ok := body[key].([]string); ok {
            for i, u := range arr {
                arr[i] = uc.inlineOneMedia(ctx, u)
            }
        }
    }
    
    // 单值字段
    for _, key := range []string{"image", "start_image", "audio_url"} {
        if u, ok := body[key].(string); ok {
            body[key] = uc.inlineOneMedia(ctx, u)
        }
    }
}

func (uc *GenerationUseCase) inlineOneMedia(ctx context.Context, u string) string {
    if u == "" || strings.HasPrefix(u, "data:") {
        return u
    }
    
    data, mime, ok := uc.asset.ReadLocal(ctx, u)
    if !ok || len(data) == 0 || len(data) > 15<<20 {
        return u  // 超 15MB 保留 URL
    }
    
    return "data:" + mime + ";base64," + base64.StdEncoding.EncodeToString(data)
}

// 任务失败后可人工重试
func (uc *GenerationUseCase) Retry(ctx context.Context, tenantID, taskID string) (entity.GenerationTask, error) {
    task, err := uc.repo.FindByID(ctx, tenantID, taskID)
    if err != nil {
        return entity.GenerationTask{}, err
    }
    
    if task.State != entity.TaskStateFailed {
        return entity.GenerationTask{}, fmt.Errorf("只能重试失败的任务")
    }
    
    // 重置状态
    task.State = entity.TaskStateCreated
    task.ErrMsg = ""
    task.RetryCount++
    task.ProviderTaskID = ""
    
    // 重新提交
    return uc.Submit(ctx, SubmitInput{
        TenantID: task.TenantID,
        BrandID:  task.BrandID,
        SubType:  task.SubType,
        Model:    task.Model,
        Params:   entity.GenerationParamsFromJSON(task.ParamsJSON),
    })
}

## 12. 总结

内容生成模块的设计体现了整洁架构的核心思想：

1. **统一协议**：所有端点共享同一任务模型，协议层只写一次
2. **策略驱动**：端点差异通过 EndpointAdapter 策略实现，新增端点零改动用例层
3. **数据库驱动**：模型能力由 generation_specs 表管理，管理后台可热更新
4. **双通道结果获取**：回调（秒级推送）+ 轮询（20s 兜底），幂等合并
5. **可扩展性**：新增服务商 = 新 provider 包 + main 装配一行
